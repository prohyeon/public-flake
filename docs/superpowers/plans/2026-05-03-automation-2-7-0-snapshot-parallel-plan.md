> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build version 2.7.0 automation that captures an initial state snapshot, runs safe independent work in bounded parallel groups, captures a final snapshot, and selectively repairs only unfinished work.

**Architecture:** Add a small snapshot/planner/runner layer around the existing workflow modules instead of rewriting every workflow. Snapshot code normalizes API state into stable keys, the planner converts state gaps into task groups, and the runner executes parallel-safe groups while keeping irreversible or FLAKE-spending tasks serial. Existing workflow functions remain the source of action behavior, with orchestration moved out of the long sequential `runAutomation()`.

**Tech Stack:** Vite 5, ESM JavaScript, vite-plugin-monkey userscript build, Tampermonkey `GM_xmlhttpRequest`/`GM_openInTab`, Node built-in `node:test` for pure unit tests.

## Important Safety Notes

- Do not run every task in parallel. FLAKE-spending tasks, reward claims, and comment posting are not safely idempotent.
- `runRouletteDraws()` and `executePrizeEntry()` both spend FLAKE. Keep them serial in the `flakeSpending` group.
- `claimRouletteExtraRewards()` must run after `runRouletteDraws()` because EXTRA eligibility depends on roulette count.
- Daily shop accumulated rewards should run after daily shop attendance claim, because daily claim may unlock accumulated rewards.
- Visit missions need a two-phase flow: open tabs in parallel, wait `visitDelay`, recapture mission snapshot, then claim rewards.
- Comments have no current authoritative status check in this codebase. The repair pass must not blindly add more comments unless local execution recorded failed attempts.
- API status can lag after visits and claims. Reconciliation should allow one delayed recapture before retrying.
- Bound API write concurrency. Use concurrency `2` for reward/mission claims and serial execution for FLAKE-spending tasks.
- Always close tabs in `finally`, even if a parallel group fails.
- Keep `stove-quest-automation.user.js` generated by `npm run build`; do not edit it by hand.

import { state } from '../state.js';
import { getAllDailyMissions, getMissionComponentIds } from '../api/missions.js';
import { getRouletteParticipationCount, getRouletteSubEventNo, getRouletteExtra, getRouletteExtraSubEventNo } from '../api/roulette.js';
import { getDailyShopRewards, getMajakDailyShopRewards } from '../api/shop.js';
import { getTotalFlakeBalance, getMonthlyFlakeTotal } from '../api/profile.js';
import { checkArticleWriteStatus } from './status.js';

function getMissionCategory(componentType, componentNo) {
    if (componentType === 'SINGLE') return 'daily';
    if (componentType === 'CONTENT1') return 'content';
    if (componentType === 'SURVEY') return 'survey';
    if (componentType === 'BANNER') return 'banner';
    if (componentType === 'ACCUMULATION') {
        return componentNo === state.missionComponents.weekly ? 'weekly' : 'attendance';
    }
    return 'other';
}

function makeEmptyCategory() {
    return {
        all: [],
        complete: [],
        receivable: [],
        incomplete: []
    };
}

export function normalizeMissionSnapshot(components = []) {
    const categories = {
        daily: makeEmptyCategory(),
        content: makeEmptyCategory(),
        weekly: makeEmptyCategory(),
        banner: makeEmptyCategory(),
        attendance: makeEmptyCategory(),
        survey: makeEmptyCategory(),
        other: makeEmptyCategory()
    };
    const byMissionNo = {};

    for (const component of components) {
        const componentNo = component.componentNo;
        const componentType = component.component_info?.component_type;
        const category = getMissionCategory(componentType, componentNo);
        const missions = component.missions || [];

        for (const mission of missions) {
            const normalized = {
                missionNo: mission.mission_no,
                componentNo,
                componentType,
                category,
                title: mission.title,
                status: mission.status,
                missionType: mission.mission_type,
                isVisitMission: mission.is_visit_mission === true,
                rewardAmount: mission.reward_amount || 0,
                buttonUrl: mission.button_url || mission.url || null
            };

            byMissionNo[normalized.missionNo] = normalized;
            categories[category].all.push(normalized);

            if (mission.status === 'COMPLETE' || mission.status === 'COMPLETED') {
                categories[category].complete.push(normalized);
            } else if (mission.status === 'RECEIVABLE') {
                categories[category].receivable.push(normalized);
            } else if (mission.status === 'INCOMPLETE') {
                categories[category].incomplete.push(normalized);
            }
        }
    }

    return { categories, byMissionNo };
}

function getTodayShopState(shopData) {
    const rewards = shopData?.value?.daily_attendances?.rewards || [];
    const today = new Date().toISOString().slice(0, 10);
    return {
        unclaimedDaily: rewards.filter(reward => reward.attendance_date === today && !reward.is_received),
        accumulated: shopData?.value?.accumulated_attendances || null
    };
}

export async function captureAutomationSnapshot(headers, deps = {}) {
    const services = {
        getMissionComponentIds,
        getAllDailyMissions,
        getRouletteParticipationCount,
        getRouletteExtra,
        getDailyShopRewards,
        getMajakDailyShopRewards,
        getTotalFlakeBalance,
        getMonthlyFlakeTotal,
        checkArticleWriteStatus,
        ...deps
    };

    if (!Object.values(state.missionComponents).some(Boolean)) {
        await services.getMissionComponentIds(headers);
    }

    const [
        articleWrite,
        allMissions,
        roulette,
        rouletteExtra,
        dailyShop,
        majakShop,
        totalFlake,
        monthlyFlake
    ] = await Promise.all([
        services.checkArticleWriteStatus(headers).catch(error => ({ success: false, error: error.message })),
        services.getAllDailyMissions(headers).catch(() => []),
        services.getRouletteParticipationCount(headers, getRouletteSubEventNo()).catch(error => ({ success: false, error: error.message })),
        services.getRouletteExtra(headers, getRouletteExtraSubEventNo()).catch(error => ({ success: false, error: error.message })),
        services.getDailyShopRewards(headers).catch(error => ({ success: false, error: error.message })),
        services.getMajakDailyShopRewards(headers).catch(error => ({ success: false, error: error.message })),
        services.getTotalFlakeBalance(headers).catch(error => ({ error: true, message: error.message })),
        services.getMonthlyFlakeTotal(headers).catch(error => ({ error: true, message: error.message }))
    ]);

    const currentRoulette = roulette?.value?.participation_cnt || 0;
    const rouletteRemaining = Math.max(0, CONFIG.roulette.maxDraws - currentRoulette);
    const extraMilestones = rouletteExtra?.value?.milestones || [];
    const extraCurrent = rouletteExtra?.value?.current_cnt || 0;

    return {
        capturedAt: Date.now(),
        articleWrite,
        missions: normalizeMissionSnapshot(allMissions),
        roulette: {
            current: currentRoulette,
            limit: CONFIG.roulette.maxDraws,
            remaining: rouletteRemaining
        },
        rouletteExtra: {
            claimable: extraMilestones.filter(milestone => extraCurrent >= milestone.milestone && milestone.received_yn === false)
        },
        shop: getTodayShopState(dailyShop),
        majak: getTodayShopState(majakShop),
        flake: {
            total: totalFlake,
            monthly: monthlyFlake
        }
    };
}

export function compareSnapshots(before, after, plan = {}) {
    const plannedMissionNos = plan.plannedMissionNos || [];
    const incompleteMissionNos = plannedMissionNos.filter(missionNo => {
        const mission = after.missions?.byMissionNo?.[missionNo];
        return mission && mission.status !== 'COMPLETE' && mission.status !== 'COMPLETED' && mission.status !== 'RECEIVABLE';
    });

    return {
        articleStillMissing: before.articleWrite?.hasWrittenToday === false && after.articleWrite?.hasWrittenToday === false,
        rouletteStillRemaining: after.roulette?.remaining > 0,
        incompleteMissionNos,
        unclaimedDailyShop: after.shop?.unclaimedDaily?.length || 0,
        unclaimedMajakShop: after.majak?.unclaimedDaily?.length || 0,
        claimableExtra: after.rouletteExtra?.claimable?.length || 0
    };
}

export function getSnapshotSummary(snapshot) {
    const missions = Object.values(snapshot.missions?.byMissionNo || {});
    return {
        articleWritten: snapshot.articleWrite?.hasWrittenToday === true,
        rouletteRemaining: snapshot.roulette?.remaining || 0,
        missions: {
            complete: missions.filter(m => m.status === 'COMPLETE' || m.status === 'COMPLETED').length,
            receivable: missions.filter(m => m.status === 'RECEIVABLE').length,
            incomplete: missions.filter(m => m.status === 'INCOMPLETE').length
        }
    };
}

export async function runTask(task) {
    try {
        const value = await task.run();
        return { id: task.id, status: 'fulfilled', value };
    } catch (reason) {
        return { id: task.id, status: 'rejected', reason };
    }
}

export async function runLimited(tasks, concurrency = 1) {
    if (!tasks.length) return [];

    const results = new Array(tasks.length);
    let nextIndex = 0;
    const workerCount = Math.min(Math.max(1, concurrency), tasks.length);

    async function worker() {
        while (nextIndex < tasks.length) {
            const currentIndex = nextIndex;
            nextIndex++;
            results[currentIndex] = await runTask(tasks[currentIndex]);
        }
    }

    await Promise.all(Array.from({ length: workerCount }, () => worker()));
    return results;
}

export async function runTaskGroups(groups, hooks = {}) {
    const allResults = [];

    for (const group of groups) {
        hooks.onGroupStart?.(group);
        const results = await runLimited(group.tasks || [], group.concurrency || 1);
        hooks.onGroupDone?.(group, results);
        allResults.push({ groupId: group.id, results });
    }

    return allResults;
}

export function flattenTaskResults(groupResults) {
    return groupResults.flatMap(group => group.results.map(result => ({
        ...result,
        groupId: group.groupId
    })));
}

function getVisitMissionNos(snapshot) {

        .filter(mission =>
            mission.category === 'daily' &&
            mission.status === 'INCOMPLETE' &&

export function buildAutomationPlan(snapshot) {
    const visitMissionNos = getVisitMissionNos(snapshot);
    const plannedMissionNos = [...visitMissionNos];

    const groups = [
        {
            id: 'setup',
            concurrency: 3,
            tasks: [
                task('setup.requiredPages', 'requiredPages'),
                task('setup.componentRefresh', 'componentRefresh'),
                task('setup.eventRefresh', 'eventRefresh')

        },
        {
            id: 'community',
            concurrency: 3,
            tasks: [
                ...(snapshot.articleWrite?.hasWrittenToday ? [] : [task('community.articleWrite', 'articleWrite')]),
                task('community.articleLikes', 'articleLikes'),
                task('community.comments', 'comments', { serialInside: true, nonAuthoritativeRepair: true })

            id: 'visits',
            concurrency: 4,

export function buildRepairPlan(diff) {
    const groups = [];

    const safeRepairs = [];
    if (diff.articleStillMissing) safeRepairs.push(task('repair.articleWrite', 'articleWrite'));
    if (diff.incompleteMissionNos?.length) safeRepairs.push(task('repair.visitMissions', 'singleVisits', { missionNos: diff.incompleteMissionNos }));
    if (diff.unclaimedDailyShop > 0) safeRepairs.push(task('repair.dailyShop', 'dailyShop'));
    if (diff.claimableExtra > 0) safeRepairs.push(task('repair.rouletteExtra', 'rouletteExtra'));

    if (safeRepairs.length > 0) {
        groups.push({ id: 'repairSafe', concurrency: 1, tasks: safeRepairs });
    }

    return { plannedMissionNos: diff.incompleteMissionNos || [], groups };
}
